cmake_minimum_required(VERSION 3.22)
project(SylarQuestGame VERSION 0.8.2 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(OpenXR REQUIRED CONFIG)
get_filename_component(SYLAR_HOST "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../native-host" ABSOLUTE)
set(SYLAR_CORE_LIBRARY "$ENV{SYLAR_CORE_LIBRARY}" CACHE FILEPATH "Privately built ARM64 core")
if(NOT EXISTS "${SYLAR_CORE_LIBRARY}")
 message(FATAL_ERROR "Set SYLAR_CORE_LIBRARY to your locally generated libsylar_dkc1.so; binaries are not distributed.")
endif()
add_library(sylar_dkc1 SHARED IMPORTED)
set_target_properties(sylar_dkc1 PROPERTIES IMPORTED_LOCATION "${SYLAR_CORE_LIBRARY}")
add_library(native_app_glue STATIC "${ANDROID_NDK}/sources/android/native_app_glue/android_native_app_glue.c")
target_include_directories(native_app_glue SYSTEM PUBLIC "${ANDROID_NDK}/sources/android/native_app_glue")
add_library(sylar_quest_game SHARED main.c closed_relief.c reconstruction_shader.c maquette_math.c status_screen.c "${SYLAR_HOST}/host_util.c")
target_include_directories(sylar_quest_game PRIVATE "${SYLAR_HOST}")
target_compile_definitions(sylar_quest_game PRIVATE XR_USE_PLATFORM_ANDROID XR_USE_GRAPHICS_API_OPENGL_ES)
target_compile_options(sylar_quest_game PRIVATE -Wall -Wextra -Wpedantic -Werror -O2)
target_link_options(sylar_quest_game PRIVATE -Wl,-u,ANativeActivity_onCreate -Wl,--no-undefined)
target_link_libraries(sylar_quest_game PRIVATE native_app_glue sylar_dkc1 OpenXR::openxr_loader android log EGL GLESv3 m)
